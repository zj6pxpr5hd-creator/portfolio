src/vm/eval.c — Dispatch Loop C99 
typedef enum { OP_PUSH, OP_ADD, OP_LOAD, OP_JMP_IF_ZERO, OP_HALT } OpCode;

int64_t vm_eval(VM *vm) {
  while (vm->pc < vm->code_len) {
    uint8_t op = vm->bytecode[vm->pc++];
    switch (op) {
      case OP_ADD: {
        int64_t b = pop(vm);
        int64_t a = pop(vm);
        push(vm, a + b); // Fast register-mapped ALU op
        break;
      }
      case OP_JMP_IF_ZERO: {
        int64_t cond = pop(vm);
        uint16_t addr = read_u16(vm);
        if (cond == 0) vm->pc = addr;
        break;
      }
      case OP_HALT: return pop(vm);
    }
  }
  return -1;
}

raft/append_entries.go — Heartbeat & Log Sync Go 1.22 
func (rf *RaftNode) AppendEntries(args *AppendArgs, reply *AppendReply) {
    rf.mu.Lock()
    defer rf.mu.Unlock()

    if args.Term < rf.currentTerm {
        reply.Success, reply.Term = false, rf.currentTerm
        return // Reject stale leader
    }
    if args.Term > rf.currentTerm {
        rf.currentTerm, rf.state = args.Term, StateFollower
        rf.votedFor = -1
    }
    rf.resetElectionTimer() // Valid leader keeps node calm
    if len(args.Entries) > 0 {
        rf.log = append(rf.log[:args.PrevLogIndex+1], args.Entries...)
    }
    reply.Success = true
}

export async function claimSlot(studentId: string, slotId: string) {
  return await db.transaction(async (tx) => {
    // Row-level lock prevents double-booking race condition
    const [slot] = await tx.query`
      SELECT id, status FROM tutor_slots 
      WHERE id = ${slotId} AND status = 'OPEN'
      FOR UPDATE SKIP LOCKED`;
      
    if (!slot) throw new Error('Slot already reserved by peer');
    await tx.query`UPDATE tutor_slots SET status = 'BOOKED', student = ${studentId}`;
    wsBroadcaster.publish('room_updates', { slotId, status: 'BOOKED' });
  });
}

void btree_split_child(BNode *parent, int idx, BNode *child) {
    BNode *sibling = bnode_create(child->is_leaf);
    sibling->key_count = T - 1;

    // Copy upper half of keys into sibling node
    for (int j = 0; j < T - 1; j++) {
        sibling->keys[j] = child->keys[j + T];
    }
    int64_t median_key = child->keys[T - 1];
    child->key_count = T - 1;

    // Promote pivot key into parent index frame
    parent_insert_pivot(parent, idx, median_key, sibling);
    wal_append_split_record(parent->page_id, child->page_id, sibling->page_id);
}
